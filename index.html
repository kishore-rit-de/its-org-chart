<script>

(function() {

    const allNodes = Array.from(document.querySelectorAll('.tree-node'));

    const searchBox = document.getElementById('searchBox');

    const status = document.getElementById('status');


    function setStatus(text) { status.textContent = text; }


    function revealAllNodes() {

        allNodes.forEach(node => {

            node.classList.remove('hidden', 'search-hit', 'collapsed-branch');

            const btn = node.querySelector(':scope > .node-card > .toggle-btn');

            if(btn) btn.textContent = '−';

        });

    }


    function expandAll() {

        revealAllNodes();

        setStatus('Expanded all branches');

    }


    function directReportsOnly() {

        searchBox.value = '';

        revealAllNodes();

        allNodes.forEach(node => {

            const level = parseInt(node.dataset.level || '0');

            if (level === 1) {

                node.classList.add('collapsed-branch');

                const btn = node.querySelector(':scope > .node-card > .toggle-btn');

                if(btn) btn.textContent = '+';

            }

        });

        setStatus('Showing VP/CIO and direct reports');

    }


    function clearSearch() {

        searchBox.value = '';

        revealAllNodes();

        setStatus('Search cleared');

    }


    function runSearch() {

        const query = searchBox.value.trim().toLowerCase();

        allNodes.forEach(node => node.classList.remove('search-hit', 'hidden', 'collapsed-branch'));

        

        if (!query) {

            setStatus('Ready');

            return;

        }


        const matching = [];

        allNodes.forEach(node => {

            const ownText = (node.dataset.text || '').toLowerCase();

            if (ownText.includes(query)) {

                matching.push(node);

                node.classList.add('search-hit');

            }

        });


        allNodes.forEach(node => node.classList.add('hidden'));

        matching.forEach(node => {

            node.classList.remove('hidden');

            let current = node.parentElement;

            while (current) {

                if (current.closest('.tree-node')) {

                    const parentNode = current.closest('.tree-node');

                    parentNode.classList.remove('hidden', 'collapsed-branch');

                    const btn = parentNode.querySelector(':scope > .node-card > .toggle-btn');

                    if(btn) btn.textContent = '−';

                }

                current = current.parentElement;

            }

        });


        setStatus(matching.length === 1 ? '1 match' : matching.length + ' matches');

    }


    // Toggle branches on node click or indicator click

    document.querySelector('.tree').addEventListener('click', function(e) {

        const toggleBtn = e.target.closest('.toggle-btn');

        const card = e.target.closest('.node-card');

        

        if (toggleBtn || card) {

            const node = (toggleBtn || card).closest('.tree-node');

            if (node && node.classList.contains('has-children')) {

                node.classList.toggle('collapsed-branch');

                const btn = node.querySelector(':scope > .node-card > .toggle-btn');

                if(btn) {

                    btn.textContent = node.classList.contains('collapsed-branch') ? '+' : '−';

                }

            }

        }

    });


    document.getElementById('expandAll').addEventListener('click', expandAll);

    document.getElementById('collapseAll').addEventListener('click', directReportsOnly);

    document.getElementById('clearSearch').addEventListener('click', clearSearch);

    searchBox.addEventListener('input', runSearch);


    // Initial view initialization

    directReportsOnly();

})();

</script>
